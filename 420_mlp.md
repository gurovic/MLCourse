# Полносвязные нейронные сети (MLP): От понимания к практике

## 🎯 Почему MLP? Мотивирующий пример

Представьте, что вы пытаетесь разделить эти два класса точек:


**Вопрос:** Может ли прямая линия разделить эти точки? Попробуйте мысленно провести любую прямую — некоторые точки всегда окажутся по "неправильную" сторону.

**Ответ:** Нет! Это **нелинейно разделимая** задача. Линейные модели (как логистическая регрессия) здесь бессильны.

**Решение:** Нужна модель, которая может "изгибаться" и создавать сложные границы. Именно это и делает MLP!

## 🏗️ Основные компоненты MLP

### 1. Линейный слой: строительный блок

Линейный слой выполняет простое преобразование:
```
выход = вход × веса + смещения
```

```python
class SimpleLinearLayer:
    """Упрощённая версия линейного слоя для понимания"""
    def __init__(self, input_size, output_size):
        self.weights = torch.randn(input_size, output_size) * 0.1
        self.biases = torch.zeros(output_size)
    
    def forward(self, x):
        return x @ self.weights + self.biases

# Пример: 3 признака на входе, 2 нейрона в слое
layer = SimpleLinearLayer(3, 2)
x_example = torch.tensor([1.0, 2.0, 3.0])
print(f"Вход: {x_example}")
print(f"Выход слоя: {layer.forward(x_example)}")
print(f"Веса слоя:\n{layer.weights}")
```

### 2. Функции активации: добавляем нелинейность

**Ключевой момент:** Без активаций цепочка линейных слоёв эквивалентна одному линейному слою!

```python
def demonstrate_linearity_problem():
    """Показываем, почему без активаций многослойность бесполезна"""
    
    # Два линейных слоя БЕЗ активации
    layer1 = nn.Linear(2, 2)
    layer2 = nn.Linear(2, 2)
    
    x = torch.tensor([[1.0, 2.0]])
    
    # Прямой проход через два слоя
    y1 = layer1(x)
    y2 = layer2(y1)
    
    # А теперь один слой, который делает то же самое!
    # y2 = x @ W1 @ W2 + b1 @ W2 + b2
    combined_weight = layer1.weight @ layer2.weight
    combined_bias = layer1.bias @ layer2.weight + layer2.bias
    
    print("Два слоя без активации дают:")
    print(f"  y2 = {y2}")
    print("\nОдин слой с параметрами W_combined, b_combined даёт:")
    print(f"  y_combined = {x @ combined_weight + combined_bias}")
    print("\nОни одинаковы! Многослойность теряет смысл.")

demonstrate_linearity_problem()
```

## 🧩 Собираем всё вместе: простая MLP

```python
class SimpleMLP(nn.Module):
    """Полносвязная сеть с настраиваемой архитектурой"""
    
    def __init__(self, input_size, hidden_sizes, output_size):
        """
        Args:
            input_size: размер входных данных
            hidden_sizes: список размеров скрытых слоёв
            output_size: размер выходных данных
        """
        super().__init__()
        
        layers = []
        prev_size = input_size
        
        # Добавляем скрытые слои
        for hidden_size in hidden_sizes:
            layers.append(nn.Linear(prev_size, hidden_size))
            layers.append(nn.ReLU())  # Активация после каждого скрытого слоя
            prev_size = hidden_size
        
        # Выходной слой (без активации для CrossEntropyLoss)
        layers.append(nn.Linear(prev_size, output_size))
        
        self.network = nn.Sequential(*layers)
    
    def forward(self, x):
        """Прямой проход через всю сеть"""
        return self.network(x)

# Пример: создаём MLP для задачи с кругами
mlp_circles = SimpleMLP(
    input_size=2,        # 2 признака (x, y координаты)
    hidden_sizes=[8, 4], # 2 скрытых слоя: 8 и 4 нейрона
    output_size=2        # 2 класса
)

print("Создана MLP со структурой:")
print(mlp_circles)
print(f"\nОбщее число параметров: {sum(p.numel() for p in mlp_circles.parameters()):,}")
```

## 🎯 Решаем задачу с кругами

```python
def solve_circle_problem():
    """Обучаем MLP разделять круговые данные"""
    
    # Подготовка данных
    X_tensor = torch.FloatTensor(X)
    y_tensor = torch.LongTensor(y)
    
    # Создаём и обучаем модель
    model = SimpleMLP(input_size=2, hidden_sizes=[16, 8], output_size=2)
    
    # Функция потерь и оптимизатор
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.01)
    
    # Простой цикл обучения
    losses = []
    accuracies = []
    
    print("Начинаем обучение...")
    for epoch in range(200):
        # Обнуляем градиенты
        optimizer.zero_grad()
        
        # Прямой проход
        predictions = model(X_tensor)
        
        # Вычисляем ошибку
        loss = criterion(predictions, y_tensor)
        
        # Обратное распространение
        loss.backward()
        
        # Обновляем веса
        optimizer.step()
        
        # Вычисляем точность
        with torch.no_grad():
            predicted_classes = predictions.argmax(dim=1)
            accuracy = (predicted_classes == y_tensor).float().mean().item()
        
        losses.append(loss.item())
        accuracies.append(accuracy)
        
        if epoch % 40 == 0:
            print(f"Эпоха {epoch:3d}: ошибка = {loss.item():.4f}, точность = {accuracy:.2%}")
    
    # Визуализируем результат
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    # График обучения
    axes[0].plot(losses, 'r-', label='Ошибка')
    axes[0].set_xlabel('Эпоха')
    axes[0].set_ylabel('Значение ошибки', color='red')
    axes[0].tick_params(axis='y', labelcolor='red')
    axes[0].grid(True, alpha=0.3)
    
    ax2 = axes[0].twinx()
    ax2.plot(accuracies, 'b-', label='Точность')
    ax2.set_ylabel('Точность', color='blue')
    ax2.tick_params(axis='y', labelcolor='blue')
    
    lines1, labels1 = axes[0].get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    axes[0].legend(lines1 + lines2, labels1 + labels2, loc='center right')
    axes[0].set_title('Процесс обучения')
    
    # Граница решений
    xx, yy = np.meshgrid(np.linspace(-3, 3, 100), np.linspace(-3, 3, 100))
    grid_points = torch.FloatTensor(np.c_[xx.ravel(), yy.ravel()])
    
    with torch.no_grad():
        model.eval()
        grid_predictions = model(grid_points)
        grid_probs = torch.softmax(grid_predictions, dim=1)
        Z = grid_probs[:, 1].reshape(xx.shape)
    
    axes[1].contourf(xx, yy, Z, levels=20, cmap='RdYlBu', alpha=0.8)
    axes[1].scatter(X[y==0, 0], X[y==1, 1], c='blue', label='Класс 0', alpha=0.7, edgecolors='black')
    axes[1].scatter(X[y==1, 0], X[y==1, 1], c='red', label='Класс 1', alpha=0.7, edgecolors='black')
    axes[1].set_xlabel('Признак 1')
    axes[1].set_ylabel('Признак 2')
    axes[1].set_title('Граница решений MLP')
    axes[1].legend()
    axes[1].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    print(f"\nИтоговая точность: {accuracies[-1]:.2%}")
    print("MLP успешно научилась разделять круговые данные!")
    
    return model

trained_model = solve_circle_problem()
```

## 🩺 Диагностика проблем обучения

Когда MLP не обучается, нужно уметь найти причину. Вот основные проблемы и способы их диагностики:

### Проблема 1: Нулевые или очень маленькие градиенты

```python
def check_gradients(model):
    """Проверяем, есть ли градиенты в модели"""
    print("Проверка градиентов:")
    print("-" * 40)
    
    has_gradients = False
    for name, param in model.named_parameters():
        if param.grad is not None:
            grad_norm = param.grad.norm().item()
            if grad_norm > 1e-7:
                has_gradients = True
                print(f"✓ {name:20} | градиент = {grad_norm:.2e}")
            else:
                print(f"⚠ {name:20} | СЛИШКОМ МАЛЫЙ градиент = {grad_norm:.2e}")
        else:
            print(f"✗ {name:20} | НЕТ ГРАДИЕНТА")
    
    if not has_gradients:
        print("\n🚨 ВНИМАНИЕ: Градиенты отсутствуют или слишком малы!")
        print("Возможные причины:")
        print("1. Learning rate слишком мал")
        print("2. Все активации 'умерли' (ReLU проблема)")
        print("3. Данные не нормализованы")
    else:
        print("\n✅ Градиенты присутствуют")

# Пример использования
test_model = SimpleMLP(2, [4, 2], 2)
```

### Проблема 2: Взрыв или исчезновение активаций

```python
def check_activations(model, sample_input):
    """Проверяем значения активаций в сети"""
    print("\nПроверка активаций:")
    print("-" * 40)
    
    activations = {}
    
    # Хук для сбора активаций
    def get_activation(name):
        def hook(model, input, output):
            activations[name] = output.detach()
        return hook
    
    # Регистрируем хуки на ReLU слоях
    hooks = []
    for name, module in model.named_modules():
        if isinstance(module, nn.ReLU):
            hook = module.register_forward_hook(get_activation(name))
            hooks.append(hook)
    
    # Прямой проход
    with torch.no_grad():
        _ = model(sample_input)
    
    # Анализируем активации
    for name, activation in activations.items():
        act_values = activation.flatten()
        print(f"{name:15} | мин = {act_values.min():7.3f} | "
              f"макс = {act_values.max():7.3f} | "
              f"среднее = {act_values.mean():7.3f}")
        
        # Проверяем на "мёртвые нейроны" (ReLU)
        if isinstance(next(iter(model.modules())), nn.ReLU):
            dead_ratio = (activation == 0).float().mean().item()
            if dead_ratio > 0.8:
                print(f"  ⚠  {dead_ratio:.1%} нейронов 'мёртвые'")
    
    # Убираем хуки
    for hook in hooks:
        hook.remove()
    
    return activations

# Пример
sample_data = torch.randn(5, 2)  # 5 примеров, 2 признака
```

```python
def analyze_weights(model):
    """Анализируем распределение весов модели"""
    print("\nАнализ весов модели:")
    print("-" * 40)
    
    weight_stats = []
    
    for name, param in model.named_parameters():
        if 'weight' in name:
            weights = param.data.cpu().numpy().flatten()
            stats = {
                'name': name,
                'mean': np.mean(weights),
                'std': np.std(weights),
                'min': np.min(weights),
                'max': np.max(weights),
                'abs_mean': np.mean(np.abs(weights))
            }
            weight_stats.append(stats)
    
    # Выводим статистику
    for stats in weight_stats:
        print(f"{stats['name']:20}")
        print(f"  Среднее: {stats['mean']:8.4f} ± {stats['std']:8.4f}")
        print(f"  Диапазон: [{stats['min']:8.4f}, {stats['max']:8.4f}]")
        print(f"  Среднее |вес|: {stats['abs_mean']:8.4f}")
        
        # Проверки
        if stats['abs_mean'] < 0.01:
            print("  ⚠  Веса слишком близки к нулю")
        if stats['std'] > 1.0:
            print("  ⚠  Слишком большой разброс весов")
        print()
```

```python
def quick_diagnostic(model, X, y):
    """Быстрая диагностика проблем обучения"""
    print("=" * 50)
    print("БЫСТРАЯ ДИАГНОСТИКА ОБУЧЕНИЯ")
    print("=" * 50)
    
    model.eval()
    
    # 1. Проверяем выходы модели
    with torch.no_grad():
        predictions = model(X)
        probs = torch.softmax(predictions, dim=1)
    
    print("\n1. Выходы модели:")
    print(f"   Predictions min/max: {predictions.min():.3f} / {predictions.max():.3f}")
    print(f"   Probabilities range: {probs.min():.3f} / {probs.max():.3f}")
    
    # 2. Проверяем, не все ли предсказания одинаковы
    pred_classes = predictions.argmax(dim=1)
    unique_classes = torch.unique(pred_classes)
    if len(unique_classes) < 2:
        print(f"   ⚠  Модель предсказывает только класс {unique_classes[0].item()}")
    
    # 3. Базовая точность
    accuracy = (pred_classes == y).float().mean().item()
    print(f"   Базовая точность: {accuracy:.2%}")
    
    if accuracy < 0.5:
        print("   ⚠  Точность ниже случайного угадывания")
    
    # 4. Проверяем потерю
    criterion = nn.CrossEntropyLoss()
    loss = criterion(predictions, y).item()
    print(f"\n2. Значение функции потерь: {loss:.4f}")
    
    # Для 2 классов, начальная loss должна быть около -ln(0.5) ≈ 0.69
    if loss < 0.1:
        print("   ⚠  Слишком маленькая потеря (возможно переобучение)")
    elif loss > 2.0:
        print("   ⚠  Слишком большая потеря")
    
    print("\n3. Рекомендации:")
    if accuracy < 0.5 and loss > 1.0:
        print("   • Увеличить learning rate")
        print("   • Проверить инициализацию весов")
        print("   • Нормализовать входные данные")
    elif accuracy > 0.9 and loss < 0.1:
        print("   • Модель хорошо обучена!")
    else:
        print("   • Продолжить обучение")
        print("   • Следить за валидационной точностью")

# Пример использования диагностики
quick_diagnostic(trained_model, X_tensor, y_tensor)
```

## 📊 Практическое руководство: когда что делать

### Если модель не обучается:

1. **Проверьте градиенты** — есть ли они вообще?
2. **Проверьте активации** — не все ли нейроны "мёртвые"?
3. **Попробуйте другой learning rate** — начните с 0.001
4. **Упростите архитектуру** — начните с 1 скрытого слоя
5. **Проверьте данные** — нет ли ошибок в разметке?

### Если обучается плохо (низкая точность):

1. **Увеличьте размер сети** — больше нейронов или слоёв
2. **Добавьте больше данных** — или аугментируйте существующие
3. **Измените функцию активации** — попробуйте LeakyReLU
4. **Обучение дольше** — увеличьте число эпох

### Если точность скачет:

1. **Уменьшите learning rate**
2. **Добавьте нормализацию данных**
3. **Используйте более стабильный оптимизатор** — Adam вместо SGD
