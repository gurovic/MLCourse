## 🟢 Базовый уровень: Понимаем основные концепции

### Что такое MLP?

MLP (Multi-Layer Perceptron) — это последовательность **линейных слоёв** с **нелинейными активациями** между ними.

#### 1. Линейный слой: самый простой блок

```python
class SimpleLinearLayer(nn.Module):
    """Простейший линейный слой для понимания"""
    def __init__(self, input_size, output_size):
        super().__init__()
        self.linear = nn.Linear(input_size, output_size)
    
    def forward(self, x):
        return self.linear(x)

# Создаём и тестируем
layer = SimpleLinearLayer(3, 2)
x_test = torch.tensor([[1.0, 2.0, 3.0]])
print(f"Вход: {x_test}")
print(f"Выход: {layer(x_test)}")
print(f"Форма весов: {layer.linear.weight.shape}")
```

**Что делает линейный слой?**
```
выход = вход × веса + смещения
```

#### 3. Собираем простую MLP

```python
class BeginnerMLP(nn.Module):
    """MLP для начинающих: 1 скрытый слой"""
    
    def __init__(self, input_size, hidden_size, output_size):
        super().__init__()
        self.layer1 = nn.Linear(input_size, hidden_size)
        self.activation = nn.ReLU()
        self.layer2 = nn.Linear(hidden_size, output_size)
    
    def forward(self, x):
        x = self.layer1(x)
        x = self.activation(x)  # Добавляем нелинейность!
        x = self.layer2(x)
        return x

# Создаём модель для задачи с кругами
simple_model = BeginnerMLP(input_size=2, hidden_size=8, output_size=2)
print("Модель создана!")
print(f"Параметры: {sum(p.numel() for p in simple_model.parameters()):,}")
```

## 🟡 Средний уровень: Обучение и оценка MLP

### 1. Полный цикл обучения

```python
class MLPTrainer:
    """Тренер для обучения MLP"""
    
    def __init__(self, model, learning_rate=0.01):
        self.model = model
        self.criterion = nn.CrossEntropyLoss()
        self.optimizer = optim.Adam(model.parameters(), lr=learning_rate)
        self.history = {'loss': [], 'accuracy': []}
    
    def train_epoch(self, X, y):
        """Одна эпоха обучения"""
        self.model.train()
        
        # Прямой проход
        predictions = self.model(X)
        loss = self.criterion(predictions, y)
        
        # Обратное распространение
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        
        # Вычисляем точность
        pred_classes = predictions.argmax(dim=1)
        accuracy = (pred_classes == y).float().mean().item()
        
        return loss.item(), accuracy
    
    def train(self, X, y, epochs=100, verbose=True):
        """Полный цикл обучения"""
        for epoch in range(epochs):
            loss, accuracy = self.train_epoch(X, y)
            self.history['loss'].append(loss)
            self.history['accuracy'].append(accuracy)
            
            if verbose and (epoch + 1) % 20 == 0:
                print(f"Эпоха {epoch+1:3d}/{epochs} | "
                      f"Loss: {loss:.4f} | Accuracy: {accuracy:.2%}")
    
    def evaluate(self, X, y):
        """Оценка модели на новых данных"""
        self.model.eval()
        with torch.no_grad():
            predictions = self.model(X)
            loss = self.criterion(predictions, y).item()
            pred_classes = predictions.argmax(dim=1)
            accuracy = (pred_classes == y).float().mean().item()
        return loss, accuracy
    
    def plot_history(self):
        """Визуализация процесса обучения"""
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 4))
        
        ax1.plot(self.history['loss'], 'r-', linewidth=2)
        ax1.set_xlabel('Эпоха')
        ax1.set_ylabel('Loss')
        ax1.set_title('Функция потерь')
        ax1.grid(True, alpha=0.3)
        
        ax2.plot(self.history['accuracy'], 'g-', linewidth=2)
        ax2.set_xlabel('Эпоха')
        ax2.set_ylabel('Accuracy')
        ax2.set_title('Точность')
        ax2.grid(True, alpha=0.3)
        
        plt.tight_layout()
        plt.show()
```

```python
def plot_decision_boundary(model, X_data, y_data):
    """Визуализация границы решений"""
    # Создаём сетку
    x_min, x_max = X_data[:, 0].min() - 0.5, X_data[:, 0].max() + 0.5
    y_min, y_max = X_data[:, 1].min() - 0.5, X_data[:, 1].max() + 0.5
    
    xx, yy = np.meshgrid(np.linspace(x_min, x_max, 100),
                         np.linspace(y_min, y_max, 100))
    
    # Предсказываем для всех точек
    grid_points = torch.FloatTensor(np.c_[xx.ravel(), yy.ravel()])
    
    model.eval()
    with torch.no_grad():
        predictions = model(grid_points)
        probabilities = torch.softmax(predictions, dim=1)
        Z = probabilities[:, 1].reshape(xx.shape)
    
    # Рисуем
    plt.figure(figsize=(10, 8))
    plt.contourf(xx, yy, Z, alpha=0.8, cmap='RdYlBu')
    plt.scatter(X_data[:, 0], X_data[:, 1], c=y_data, 
                edgecolors='black', cmap='RdYlBu')
    plt.title('Граница решений MLP')
    plt.xlabel('Признак 1')
    plt.ylabel('Признак 2')
    plt.colorbar(label='Вероятность класса 1')
    plt.grid(True, alpha=0.3)
    plt.show()

def solve_circle_problem_intermediate():

        nn.ReLU(),
        nn.Linear(16, 8),   # 16 → 8 нейронов

    print(f"\nИтоговая точность: {final_acc:.2%}")
    
    # Визуализируем границу решений
    plot_decision_boundary(model, X, y)
    
    return model, trainer
```

### 1. Диагностика проблем обучения

```python
class MLPDiagnostic:
    """Инструменты для диагностики MLP"""
    
    @staticmethod
    def check_gradient_flow(model, X_sample, y_sample):
        """Анализ потока градиентов"""
        print("=" * 60)
        print("АНАЛИЗ ГРАДИЕНТОВ")
        print("=" * 60)
        
        # Выполняем один шаг обучения
        criterion = nn.CrossEntropyLoss()
        predictions = model(X_sample)
        loss = criterion(predictions, y_sample)
        loss.backward()
        
        # Анализируем градиенты
        grad_stats = []
        for name, param in model.named_parameters():
            if param.grad is not None:
                grad_norm = param.grad.norm().item()
                grad_stats.append((name, grad_norm))
                
                status = "✓ НОРМА"
                if grad_norm < 1e-7:
                    status = "⚠ МАЛЕНЬКИЙ"
                elif grad_norm > 100:
                    status = "⚠ БОЛЬШОЙ"
                
                print(f"{name:25} | norm = {grad_norm:10.2e} | {status}")
            else:
                print(f"{name:25} | НЕТ ГРАДИЕНТА")
        
        return grad_stats
    
    @staticmethod
    def analyze_activations(model, X_sample):
        """Анализ распределения активаций"""
        print("\n" + "=" * 60)
        print("АНАЛИЗ АКТИВАЦИЙ")
        print("=" * 60)
        
        activations = {}
        
        def hook_fn(name):
            def hook(module, input, output):
                activations[name] = output.detach()
            return hook
        
        # Регистрируем хуки на ReLU слоях
        hooks = []
        for name, module in model.named_modules():
            if isinstance(module, nn.ReLU):
                hook = module.register_forward_hook(hook_fn(name))
                hooks.append(hook)
        
        # Прямой проход
        with torch.no_grad():
            _ = model(X_sample)
        
        # Анализируем
        for name, activation in activations.items():
            act_flat = activation.flatten()
            mean_val = act_flat.mean().item()
            std_val = act_flat.std().item()
            zero_ratio = (activation == 0).float().mean().item()
            
            print(f"{name:20}")
            print(f"  Среднее: {mean_val:8.4f} ± {std_val:8.4f}")
            print(f"  Нулевых: {zero_ratio:7.1%}")
            
            if zero_ratio > 0.8:
                print("  ⚠  МНОГО 'МЁРТВЫХ' НЕЙРОНОВ")
            if std_val > 5.0:
                print("  ⚠  БОЛЬШОЙ РАЗБРОС АКТИВАЦИЙ")
        
        # Убираем хуки
        for hook in hooks:
            hook.remove()
        
        return activations
    
    @staticmethod
    def check_weight_distribution(model):
        """Анализ распределения весов"""
        print("\n" + "=" * 60)
        print("АНАЛИЗ ВЕСОВ")
        print("=" * 60)
        
        weight_info = []
        for name, param in model.named_parameters():
            if 'weight' in name:
                weights = param.data.cpu().numpy().flatten()
                info = {
                    'name': name,
                    'mean': np.mean(weights),
                    'std': np.std(weights),
                    'abs_mean': np.mean(np.abs(weights))
                }
                weight_info.append(info)
                
                print(f"{name:20}")
                print(f"  μ = {info['mean']:8.4f} | σ = {info['std']:8.4f} | |μ| = {info['abs_mean']:8.4f}")
                
                if info['abs_mean'] < 0.01:
                    print("  ⚠  ВЕСА БЛИЗКИ К НУЛЮ")
                if info['std'] > 1.0:
                    print("  ⚠  БОЛЬШОЙ РАЗБРОС ВЕСОВ")
        
        return weight_info
```

### 2. Продвинутая архитектура с остаточными связями

```python
class ResidualBlock(nn.Module):
    """Остаточный блок для глубоких сетей"""
    
    def __init__(self, hidden_size):
        super().__init__()
        self.block = nn.Sequential(
            nn.Linear(hidden_size, hidden_size),
            nn.ReLU(),
            nn.Linear(hidden_size, hidden_size),
        )
        self.activation = nn.ReLU()
    
    def forward(self, x):
        residual = x
        out = self.block(x)
        out += residual  # Остаточная связь!
        out = self.activation(out)
        return out

class AdvancedMLP(nn.Module):
    """Продвинутая MLP с остаточными связями"""
    
    def __init__(self, input_size, hidden_size, num_blocks, output_size):
        super().__init__()
        
        # Входной слой
        self.input_layer = nn.Sequential(
            nn.Linear(input_size, hidden_size),
            nn.ReLU()
        )
        
        # Остаточные блоки
        self.res_blocks = nn.Sequential(*[
            ResidualBlock(hidden_size) for _ in range(num_blocks)
        ])
        
        # Выходной слой
        self.output_layer = nn.Linear(hidden_size, output_size)
    
    def forward(self, x):
        x = self.input_layer(x)
        x = self.res_blocks(x)
        x = self.output_layer(x)
        return x

# Создаём глубокую сеть
deep_model = AdvancedMLP(
    input_size=2,
    hidden_size=32,
    num_blocks=5,      # 5 остаточных блоков = 10 линейных слоёв!
    output_size=2
)

print("Глубокая остаточная сеть создана!")
print(f"Общее число параметров: {sum(p.numel() for p in deep_model.parameters()):,}")
```
