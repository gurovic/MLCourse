# Полносвязные нейронные сети (MLP): От основ к практике

## 🎯 Мотивирующий пример

```python
import numpy as np
import torch
import torch.nn as nn

# Создаём нелинейно разделимые данные
np.random.seed(42)
X = np.random.randn(300, 2)
y = ((X[:, 0]**2 + X[:, 1]**2) > 1.5).astype(int)
```

**Проблема:** Эти точки нельзя разделить прямой линией.  
**Решение:** Нужна модель, которая может "изгибаться" — MLP!

## 🟢 Базовый уровень

### 1. Линейный слой — самый простой блок

```python
class LinearLayer(nn.Module):
    def __init__(self, input_size, output_size):
        super().__init__()
        self.linear = nn.Linear(input_size, output_size)
    
    def forward(self, x):
        return self.linear(x)

# Тестируем
layer = LinearLayer(3, 2)
x_test = torch.tensor([[1.0, 2.0, 3.0]])
print(f"Вход: {x_test}")
print(f"Выход: {layer(x_test)}")
print(f"Форма весов: {layer.linear.weight.shape}")
```

Что делает линейный слой?  
```
выход = вход × веса + смещения
```

### 2. Функции активации — "изгибаем" прямые линии

```python
x = torch.linspace(-3, 3, 100)
relu = nn.ReLU()(x)      # ReLU(x) = max(0, x)
sigmoid = torch.sigmoid(x)  # σ(x) = 1 / (1 + e^(-x))
tanh = torch.tanh(x)     # tanh(x) = (e^x - e^(-x)) / (e^x + e^(-x))
```

**Без активаций** MLP была бы просто линейной моделью!

### 3. Собираем простую MLP

```python
class SimpleMLP(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super().__init__()
        self.layer1 = nn.Linear(input_size, hidden_size)
        self.activation = nn.ReLU()
        self.layer2 = nn.Linear(hidden_size, output_size)
    
    def forward(self, x):
        x = self.layer1(x)
        x = self.activation(x)  # Добавляем нелинейность!
        x = self.layer2(x)
        return x

# Создаём модель
model = SimpleMLP(input_size=2, hidden_size=8, output_size=2)
test_input = torch.randn(1, 2)
print(f"Вход: {test_input.shape}")
print(f"Выход: {model(test_input).shape}")
print(f"Всего параметров: {sum(p.numel() for p in model.parameters()):,}")
```

## 🟡 Средний уровень

### 1. Полный цикл обучения

```python
# Подготовка данных
X_tensor = torch.FloatTensor(X)
y_tensor = torch.LongTensor(y)

# Выбор функции потерь и оптимизатора
criterion = nn.CrossEntropyLoss()  # Для классификации
optimizer = torch.optim.Adam(model.parameters(), lr=0.01)

# Цикл обучения
for epoch in range(100):
    # Прямой проход
    predictions = model(X_tensor)
    loss = criterion(predictions, y_tensor)
    
    # Обратное распространение
    optimizer.zero_grad()  # Обнуляем градиенты
    loss.backward()        # Вычисляем градиенты
    optimizer.step()       # Обновляем веса
    
    # Оценка точности
    if epoch % 20 == 0:
        pred_classes = predictions.argmax(dim=1)
        accuracy = (pred_classes == y_tensor).float().mean()
        print(f"Эпоха {epoch}: Loss={loss.item():.4f}, Accuracy={accuracy:.2%}")
```

## 🔴 Продвинутый уровень

### 1. Диагностика проблем обучения

```python
def check_gradient_flow(model, X_sample, y_sample):
    """Анализ потока градиентов"""
    criterion = nn.CrossEntropyLoss()
    
    # Один шаг обучения для получения градиентов
    predictions = model(X_sample)
    loss = criterion(predictions, y_sample)
    loss.backward()
    
    # Анализируем градиенты
    for name, param in model.named_parameters():
        if param.grad is not None:
            grad_norm = param.grad.norm().item()
            status = "✓ НОРМА"
            if grad_norm < 1e-7:
                status = "⚠ МАЛЕНЬКИЙ"
            elif grad_norm > 100:
                status = "⚠ БОЛЬШОЙ"
            print(f"{name:25} | norm = {grad_norm:10.2e} | {status}")

# Создаём тестовые данные для диагностики
X_sample = X_tensor[:10]
y_sample = y_tensor[:10]
check_gradient_flow(model, X_sample, y_sample)
```

### 2. Остаточные связи для глубоких сетей

```python
class ResidualBlock(nn.Module):
    """Остаточный блок для глубоких сетей"""
    
    def __init__(self, hidden_size):
        super().__init__()
        self.block = nn.Sequential(
            nn.Linear(hidden_size, hidden_size),
            nn.ReLU(),
            nn.Linear(hidden_size, hidden_size),
        )
        self.activation = nn.ReLU()
    
    def forward(self, x):
        residual = x
        out = self.block(x)
        out += residual  # Остаточная связь!
        out = self.activation(out)
        return out

class AdvancedMLP(nn.Module):
    """Продвинутая MLP с остаточными связями"""
    
    def __init__(self, input_size, hidden_size, num_blocks, output_size):
        super().__init__()
        
        # Входной слой
        self.input_layer = nn.Sequential(
            nn.Linear(input_size, hidden_size),
            nn.ReLU()
        )
        
        # Остаточные блоки
        self.res_blocks = nn.Sequential(*[
            ResidualBlock(hidden_size) for _ in range(num_blocks)
        ])
        
        # Выходной слой
        self.output_layer = nn.Linear(hidden_size, output_size)
    
    def forward(self, x):
        x = self.input_layer(x)
        x = self.res_blocks(x)
        x = self.output_layer(x)
        return x

# Создаём глубокую остаточную сеть
deep_model = AdvancedMLP(
    input_size=2,
    hidden_size=32,
    num_blocks=3,
    output_size=2
)

print(f"Глубокая остаточная сеть создана!")
print(f"Параметров: {sum(p.numel() for p in deep_model.parameters()):,}")
```

### 3. Эксперимент со сложными данными

```python
# Создаём сложные нелинейные данные
X_complex = np.random.randn(1000, 2)
y_complex = ((np.sin(X_complex[:, 0] * 2) + 
              X_complex[:, 0] * X_complex[:, 1] * 0.5) > 0).astype(int)

X_complex_tensor = torch.FloatTensor(X_complex)
y_complex_tensor = torch.LongTensor(y_complex)

# Тестируем разные архитектуры на сложных данных
architectures = {
    'Мелкая (3 слоя)': shallow,
    'Глубокая (6 слоёв)': deep,
    'Остаточная (3 блока)': deep_model
}

# Обучаем и сравниваем
for name, model in architectures.items():
    optimizer = torch.optim.Adam(model.parameters(), lr=0.01)
    
    # Краткое обучение
    for epoch in range(50):
        optimizer.zero_grad()
        predictions = model(X_complex_tensor)
        loss = nn.CrossEntropyLoss()(predictions, y_complex_tensor)
        loss.backward()
        optimizer.step()
    
    # Оценка
    with torch.no_grad():
        final_pred = model(X_complex_tensor)
        accuracy = (final_pred.argmax(dim=1) == y_complex_tensor).float().mean()
    
    print(f"{name}: Точность = {accuracy:.2%}")
```

## 📚 Резюме

### Минимальный рабочий пример MLP

```python
# Самая компактная реализация
class MLP(nn.Module):
    def __init__(self, layers):
        super().__init__()
        self.layers = nn.ModuleList()
        for i in range(len(layers)-1):
            self.layers.append(nn.Linear(layers[i], layers[i+1]))
            if i < len(layers)-2:
                self.layers.append(nn.ReLU())
    
    def forward(self, x):
        for layer in self.layers:
            x = layer(x)
        return x

# Создание и использование
model = MLP([2, 16, 8, 2])  # 2 входа, 2 скрытых слоя, 2 выхода
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters())

# Один шаг обучения
optimizer.zero_grad()
predictions = model(X_tensor)
loss = criterion(predictions, y_tensor)
loss.backward()
optimizer.step()
```

### Ключевые выводы

1. **MLP решает нелинейные задачи** благодаря функциям активации
2. **Глубина сети** увеличивает способность к обучению сложных зависимостей
3. **Остаточные связи** помогают обучать очень глубокие сети
4. **Диагностика градиентов** критически важна для успешного обучения
5. **Практика** — лучший способ понять работу MLP

### Быстрые советы

- **Для начала:** используйте архитектуру `[2 → 8 → 2]` с learning rate 0.01
- **Для средних задач:** попробуйте `[2 → 16 → 8 → 2]` 
- **Для сложных задач:** экспериментируйте с остаточными блоками
- **Всегда проверяйте:** градиенты и точность на каждой эпохе
